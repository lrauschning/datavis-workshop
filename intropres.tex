\begin{frame}{Why do we visualize data?}
	When we visualize data, \\
	we do this to {\bf communicate} it
\end{frame}

\begin{frame}{Visual Communication}
	Thinking of plotting as a form of communication allows us to draw an analogy to languages.\\
	A \gray{formal} language can be fully described by specifying their \emph{grammar} and \emph{lexicon}.
\end{frame}

\begin{frame}{Grammar of Graphics}
	Why do we care to do this?\\
	Formally defining a grammar allows us to
	\pause
	\begin{itemize}
		\item Describe plots systematically
		\item Understand plots we see
		\item Flexibly modify existing or invent new plots
		\item Translate our intentions to a plotting engine
	\end{itemize}
	\pause
	It does not provide advice for making beautiful or useful plots!
\end{frame}

% next pres
% explain different layers of the grammars
% go into ggplot API 

\begin{frame}{Grammar of Graphics: Components}
	In the (layered) grammar of graphics, a plot consists of:
	\begin{itemize}
		\item Data, (Transformations)
		\item Aesthetics/Mappings
		\item Geometries
		\item Stats
		\item Scales
		\item Coordinates
	\end{itemize}
	\pause
	These components can be swapped or altered individually, and composed to make a plot.
	In practice, many of these have sensible default values.
\end{frame}

\begin{frame}{Data}
	\begin{itemize}
		\item \emph{Base} of the plot
		\item A plot may integrate multiple datasets
	\end{itemize}
\end{frame}

\begin{frame}{Stat(istics)}
	\begin{itemize}
		\item Usually aggregates data
		\item May output new/other values
		\gray{\item Has a default associated geom}
			\pause
		\item Examples:
			\pause
			\begin{itemize}
				\item Identity
				\item Count
				\item Mean/Median
				\item Smoothing, Regressions
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Geom(etry)}
	\begin{itemize}
		\item Graphical primitives the plot is constructed from
		\item May have different modifiable attributes
		\gray{\item Has a default stat}
			\pause
		\item Examples:
			\pause
			\begin{itemize}
				\item Points
				\item Line
				\item Bars
				\item Boxes
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Scales}
	\begin{itemize}
		\item Describes the \emph{how} of mapping variables to aesthetics
		\item Controls of guides
		\item Can be discrete/continuous, in number/colour/other spaces
			\pause
		\item Examples:
			\pause
			\begin{itemize}
				\item Linear
				\item Log
				\item Various colourscales
				\item Binning
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Coordinates}
	\begin{itemize}
		\item Maps scale values onto plotting coordinates
		\item Examples:
			\pause
			\begin{itemize}
				\item Cartesian, 2D/3D
				\item Polar
			\end{itemize}
		\item Commonly cartesian, polar, 3D
	\end{itemize}
\end{frame}

\begin{frame}{Aesthetics/Mappings}
	\begin{itemize}
		{\bf \item Which variable gets plotted how after which transformation, and how?}
		\item \emph{Glue} of ggplot
	\end{itemize}
\end{frame}

\section*{How do these components combine to make a plot?}

\begin{frame}{Assembling a ggplot}
	\begin{itemize}
		\item A plot may be multiple layers
		\item Each layer has one data, stat, geom and aesthetics
		\item These pass information along to render the plot
			\pause
		\item In practice, defaults often work well
	\end{itemize}
\end{frame}
